    local paidKeywords = {"product", "gamepass", "robux", "passid", "devproduct"}
    local paidCache = setmetatable({}, {__mode = "k"})
    local freeCache = setmetatable({}, {__mode = "k"})

    local function hasPaidWord(str)
        str = string.lower(str)
        for _, k in ipairs(paidKeywords) do
            if string.find(str, k, 1, true) then return true end
        end
        return false
    end

    local function checkInstance(inst)
        if hasPaidWord(inst.Name) then return true end
        for attr, value in pairs(inst:GetAttributes()) do
            if hasPaidWord(attr) then return true end
            if type(value) == "string" and hasPaidWord(value) then return true end
        end
        if inst:IsA("StringValue") and hasPaidWord(inst.Value) then return true end
        if inst:IsA("TextLabel") or inst:IsA("TextButton") then
            local t = string.lower(inst.Text)
            if string.find(t, "robux", 1, true) or string.find(t, "r$", 1, true) or string.find(inst.Text, "\u{E002}", 1, true) then
                return true
            end
        end
        return false
    end

    local function isPaidButton(button)
        if paidCache[button] then return true end

        local settings = button:FindFirstChild("buttonSettings") or button:FindFirstChild("ButtonSettings")
        if settings and (settings:FindFirstChild("Product") or settings:FindFirstChild("Gamepass")) then
            paidCache[button] = true
            return true
        end

        local checkedAt = freeCache[button]
        if checkedAt and os.clock() - checkedAt < 10 then return false end

        local found = checkInstance(button)
        if not found then
            for _, d in ipairs(button:GetDescendants()) do
                if checkInstance(d) then found = true; break end
            end
        end

        if found then
            paidCache[button] = true
        else
            freeCache[button] = os.clock()
        end
        return found
    end

    local function canAffordUpgrade(button)
        local leaderstats = LP:FindFirstChild("leaderstats")
        local cash = leaderstats and leaderstats:FindFirstChild("Cash")
        local settings = button and (button:FindFirstChild("buttonSettings") or button:FindFirstChild("ButtonSettings"))
        local price = settings and settings:FindFirstChild("Price")

        if not cash or not price then return false end

        local cashValue = tonumber(cash.Value)
        local priceValue = tonumber(price.Value)
        if not cashValue or not priceValue then return false end

        -- Só permite o touchinterest quando o jogador tem Cash suficiente.
        return cashValue >= priceValue
    end

    local function tryTeleportToFreeTycoon()
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local tycoonFolder = workspace:FindFirstChild("Tycoon")
        if not hrp or not tycoonFolder then return false end

        for _, tycoon in ipairs(tycoonFolder:GetChildren()) do
            if checkTycoonFree(tycoon) then
                local things = tycoon:FindFirstChild("tycoonThings")
                local door = things and things:FindFirstChild("tycoonDoor")
                if door then
                    local targetPos
                    if door:IsA("BasePart") then
                        targetPos = door.Position
                    elseif door:IsA("Model") then
                        local ok, pivot = pcall(function() return door:GetPivot().Position end)
                        if ok then targetPos = pivot end
                    end
                    if targetPos then
                        hrp.CFrame = CFrame.new(targetPos + Vector3.new(0, 5, 0))
                        return true
                    end
                end
            end
        end
        return false
    end

    local function makeDraggable(handle, target, onTap)
        local dragging, dragStart, startPos, moved
        handle.InputBegan:Connect(function(input)
            pcall(function()
                if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    moved = false
                    dragStart = input.Position
                    startPos = target.Position
                    input.Changed:Connect(function()
                        if input.UserInputState == Enum.UserInputState.End then
                            dragging = false
                            if onTap and not moved then pcall(onTap) end
                        end
                    end)
                end
            end)
        end)
        UIS.InputChanged:Connect(function(input)
            pcall(function()
                if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                    local delta = input.Position - dragStart
                    if delta.Magnitude > 6 then moved = true end
                    target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
                end
            end)
        end)
    end

        if t then t.Completed:Wait() end
        main.Visible = false
        pill.Position = UDim2.new(0.5, -85, 0, 8)
        pillScale.Scale = 0.5
        pill.Visible = true
        tween(pillScale, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1})
    end
